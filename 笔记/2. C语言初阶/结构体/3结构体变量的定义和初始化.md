# 定义创建变量
```
struct Peo
{
	char name[10];
	char tel[12];
	char sex[5];
	int high;
}p1,p2;
```
在声明Peo结构体时,直接定义p1,p2变量,且p1,p2为全局变量
# 初始化
## 常规
```
int main()
{
	struct Peo p1 = { "张三","12345678901" ,"男",181};
	//创建Peo变量p1,并初始化
	return 0;
}
```
## 嵌套初始化
```
struct Peo
{
	char name[10];
	char tel[12];
	char sex[5];
	int high;
};

struct Stu
{
	struct Peo p1;
	int num;
	float f;
};

int main()
{
	struct Stu p2 = {{ "张三","12345678901" ,"男",181} , 123 , 100.0f};
	//创建Stu变量p2,并初始化
	printf("%s,%s,%s,%d,%d,%f\n", 
		p2.p1.name, p2.p1.tel, p2.p1.sex, p2.p1.high,
		p2.num, p2.f);
	//打印
	return 0;
}
``` 
# 打印
传址打印在结构体数据庞大时,因为不用创建临时变,所以打印更省空间
## 直接打印
```
struct Peo
{
	char name[10];
	char tel[12];
	char sex[5];
	int high;
};

struct Stu
{
	struct Peo p1;
	int num;
	float f;
};

int main()
{
	struct Stu p2 = {{ "张三","12345678901" ,"男",181} , 123 , 100.0f};
	//创建Stu变量p2,并初始化
	printf("%s,%s,%s,%d,%d,%f\n", 
		p2.p1.name, p2.p1.tel, p2.p1.sex, p2.p1.high,
		p2.num, p2.f);
	//打印
	return 0;
}
```
## 传值打印
```
struct Peo
{
	char name[10];
	char tel[12];
	char sex[5];
	int high;
};
struct Stu
{
	struct Peo p1;
	int num;
	float f;
};

void Printf(struct Stu p)
//定义打印函数
{
	printf("%s,%s,%s,%d,%d,%f\n", p.p1.name, p.p1.tel, p.p1.sex, p.p1.high, p.num, p.f);
	
}


int main()
{
	struct Stu p2 = {{ "张三","12345678901" ,"男",181} , 123 , 100.0f};
	//创建Stu变量p2,并初始化
	Printf(p2);
	//打印
	return 0;
}
```
## 传址打印
```
struct Peo
{
	char name[10];
	char tel[12];
	char sex[5];
	int high;
};
struct Stu
{
	struct Peo p1;
	int num;
	float f;
};

void Printf(struct Stu* s)
//定义打印函数
{
	printf("%s,%s,%s,%d,%d,%f\n",s->p1.name, s->p1.tel, s->p1.sex, s->p1.high,s->num, s->f);
	// ->取出地址的成员变量打印
}


int main()
{
	struct Stu p2 = {{ "张三","12345678901" ,"男",181} , 123 , 100.0f};
	//创建Stu变量p2,并初始化
	Printf(&p2);
	//打印
	return 0;
}
```
