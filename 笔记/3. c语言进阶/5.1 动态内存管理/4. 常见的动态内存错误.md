# 对NULL指针的解引用操作
```
void test()
{
	int *p = (int*)malloc(INT_MAX/4);
	*p = 20
	//如果p的值是NULL,就会有问题
	free(p);
}
```
# 对动态开辟空间的越界访问
```
int main() {
	int i = 0;
	int* p = (int*)malloc(10 * sizeof(int));
	if (NULL == p)
	{
		exit(EXIT_FAILURE);
	}
	for ( i = 0; i <= 10; i++)//当p是10的时候,越界访问
	{
		*(p + 1) = i;
	}
	free(p);
	return 0;
}
```
# 对非法动态开辟内存使用free释放
```
int main() {
	int a = 10;
	int* p = &a;
	free(p);
	//p不是动态开辟的的内存
	return 0;
}
```
# 使用free释放一块动态开辟内存的一部分
```
int main() {
	int* p = (int*)malloc(100);
	if (NULL == p)
	{
		return 1;
	}
	int i = 0;
	for ( i = 0; i < 10; i++)
	{
		*p = i; 
		p++;
	}
	free(p);//p不在指向动态内存的起始位置
	p = NULL;
	return 0;
}
```
# 对同一块动态内存多次释放
```
int main() {
	int* p = (int*)malloc(100);
	//..
	free(p);
	p = NULL;//加上p = NULL后,free(p)将不会有任何影响
	//..
	free(p);//多次释放
	return 0;
}
```
