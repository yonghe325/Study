# 柔性数组
有利于
+ 减少内存碎片
+ 连续的数组会提高访问速度
```
	typedef struct st_type {
	   int i;
	   int a[]; // 柔性数组成员
	}My_arr;
//声明结构体变量

	My_arr* pm = (My_arr*)malloc(sizeof(My_arr) + 40);
//开辟空间
	pm->i = 1200;
	for (size_t i = 0; i < 10; i++)
	{
		pm->arr[i] = i;
	}
	for (size_t i = 0; i < 10; i++)
	{
		printf("%d ", pm->arr[i]);
	}
	printf("\n");
	
	My_arr* mid =  (My_arr*)realloc(pm, sizeof(My_arr) + sizeof(int) * 20);
//动态调整数组
	if (mid !=NULL)
	{
		pm = mid;
		mid == NULL;
	}
//判断是否调整成功
	for (size_t i = 0; i < 20; i++)
	{
		pm->arr[i] = i;
	}
	for (size_t i = 0; i < 20; i++)
	{
		printf("%d ", pm->arr[i]);
	}

	free(pm);
//释放空间
```
# 以结构体成员为int\*的形式实现柔性数组
相比柔性数组方法
可能会造成
+ 内存泄露
+ 内存碎片
```
	typedef struct st_type {
	   int n;
	   int *arr; // 柔性数组成员
	}My_arr;
//声明结构体变量

	My_arr* p = (My_arr*)malloc(sizeof(My_arr));
//为结构体开辟空间
	if (p == NULL)
	{
		return 1;
	}
//判断结构体空间是否开辟成功
	p->n = 100;
	p->arr = (int*)malloc(sizeof(int) * 10);
//为数组开辟空间
	if (p->arr ==NULL)
	{
		return 1;
	}
//判断数组是否开辟成功
	for (size_t i = 0; i < 10; i++)
	{
		p->arr[i] = i;
	}
	for (size_t i = 0; i < 10; i++)
	{
		printf("%d ", p->arr[i]);
	}
//赋值,打印
	int* ptr = (int*)realloc(p->arr , sizeof(int) * 20);
//扩容数组
	if (ptr == NULL)
	{
		return 1;
	}
	p->arr = ptr;
	ptr = NULL;
//判断扩容是否成功
	for (size_t i = 0; i < 20; i++)
	{
		p->arr[i] = i;
	}
	for (size_t i = 0; i < 20; i++)
	{
		printf("%d ", p->arr[i]);
	}
//赋值,打印
```
以该方式实现柔性数组时
由于
+ 结构体使用malloc开辟了一次空间
+ 结构体内int\* 变量使用使用malloc开辟了一次空间

==当释放空间时,需要释放两次空间==
```
free(p);
free(p->arr);
p = NULL;
```
