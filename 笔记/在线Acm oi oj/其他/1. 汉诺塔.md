//有a,b,c三根柱子
//1.先将最底下的一个盘子移到目标柱子
	//将除最底下的一个盘子以外的盘子移到另一个柱子
//2.递归处理所有的盘子
#include<iostream>
using namespace std;
static int num = 0;
void move(char first, char end,int n) {
	cout << n << " from " << first << " to " << end ;
	num++;
	cout << " 第" << num << "次移动" << endl;
}
void hanno(int n,char a,char b,char c) {

	if (!(n)) return;
	hanno(--n, a, c, b);
	//移动上面的盘子
 	move(a,c,n+1);

	hanno(n,b,a,c);
	//移动上面的盘子之后,移动下面的盘子
	//最大的盘子只会移动1次
	//倒数第二大的盘子会移动2次,依次类推
	//倒数第三大的盘子会移动4次,依次类推
	//假设有n个盘子,则总共会移动(2^n)-1次
}

int main() {
	char a = 'a';
	char b = 'b';
	char c = 'c';
	int n;
	cin >> n;
	hanno(n, a, b, c);
	return 0;
}
