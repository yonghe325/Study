# 解法一 : 利用差分数组
```c++
#include<iostream>
using namespace std;
const int N = 1e5 + 10;
int  arr[N] = {0};
int f[N] = { 0 };
int main() {
	int n, m;
	cin >> n >> m;
	for (size_t i = 1; i <= n; i++) {
		cin >> arr[i];
		//获取原数组
		f[i] = arr[i] - arr[i - 1];
		//统计差分
	}
	
	int l, r, k;
	for (int i = 0; i < m; i++){
		cin >> l >> r >> k;
		f[l] += k;
		f[r + 1] -= k;
	}
	//进行修改

	//拷贝至原数组
	for (int i = 1; i <= n; i++)
	{
		cout << arr[i] << " ";
	}
	cout << endl;
	return 0;
}
```
