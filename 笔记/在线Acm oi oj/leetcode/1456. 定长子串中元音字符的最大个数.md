# 使用滑动窗口
```c++
#define MAX(a, b) ((b) > (a) ? (b) : (a))
int maxVowels(char* a, int k) {
//定长滑动窗口
//元素的移入移出
    int ans =0;
    int val =0;
    int left =0;
    for(int i =0;a[i];i++){
    //这里直接使用a[i]来判断是否越界
    //如果a[i]是数组元素, 则可以进行
    //如果越界, 则跳出循环
        if(a[i]=='a'||a[i]=='e'||a[i]=='i'||a[i]=='o'||a[i]=='u'){
            val++;
        }
        //如果是元音, val++
        left = i-k+1;
        if(left<0){
            continue;
        }
        //left根据right的位置设定
        //当left<0,即right与a[0]的位置小于k时
        //left与right的区间元素个数小于k
        //因此跳过此次循环,执行下一次
        //直到left进入
        ans = MAX(ans,val);
        //比较 当前子串(val) 与 记录中子串(ans) 中元音个数的最大值
        if(ans == k){
            break;
        }
        //若ans == k ,则表示ans已到达最大值
        //直接跳出循环
        if(a[left]=='a'||a[left]=='e'||a[left]=='i'||a[left]=='o'||a[left]=='u'){
        }
        //left将要移出,所以left的字符如果为元音,则val--
    }
    return ans;
}
```
left控制元素的移出
