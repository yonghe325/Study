遍历两次
```cpp
class Solution {
public:
    long long numberOfRightTriangles(vector<vector<int>>& grid) {
        vector<int>row(grid.size(),0);
        vector<int>col(grid[0].size(),0);
        for(int i =0; i<grid.size();i++){
            for(int j =0; j<grid[i].size();j++){
                col[j]+=grid[i][j];
                row[i]+=grid[i][j];
            }
        }
        //存储所有行列1的个数
        long long ret = 0;
        for(int i =0; i<grid.size();i++){
            for(int j =0; j<grid[i].size();j++){
                if(grid[i][j]){
                    ret += (row[i]-1) * (col[j]-1);
                }
            }
        }
        return ret;
    }
};
//存储每行每列1的个数
//对于每一个1, 三角形个数= 行的个数-1 * 列的个数-1
```
# 灵神
+ 提前-1 : 
	如果当前行列1的个数为1, 最后行列1的和即为0, 表示当前行列可以组成0个三角形的边
	如果当前行列1的个数为2, 最后行列1的和即为1, 表示当前行列可以组成1个三角形的边
	相当于把$三角形个数= 行的个数-1 * 列的个数-1$中的`-1`
	放到存储的时候, 最后计算的时候不用再计算
	
```cpp
class Solution {
public:
    long long numberOfRightTriangles(vector<vector<int>>& grid) {
        int n = grid[0].size();
        vector<int> col_sum(n, -1); // 提前减一
        for (auto& row : grid) {
            for (int j = 0; j < n; j++) {
                col_sum[j] += row[j];
            }
        }
        //统计该列的1的个数
  
        long long ans = 0;
        for (auto& row : grid) {
            int row_sum = reduce(row.begin(), row.end()) - 1; // 提前减一
            //统计改行的一的个数
            for (int j = 0; j < row.size(); j++) {
                if (row[j] == 1) {
                    ans += row_sum * col_sum[j];
                    //统计个数
                }
            }
        }
        return ans;
    }
};
```
