class Solution {
public:
    vector<int> successfulPairs(vector<int>& spells, vector<int>& potions, long long success) {
        sort(potions.begin(),potions.end());
        //先排序
        vector<int>pairs;
        for(int i : spells){
            int L = 0, R = potions.size();
            int mid = 0;
            while(L<R){
                mid = L+(R-L)/2;
                if((long long)potions[mid]*i<success){
                    L = mid+1;
                }
                else{
                    R = mid;
                }
            }
            pairs.push_back(potions.size()-L);
        }
        return pairs;
    }
};
