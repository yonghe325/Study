```cpp
class Solution {
public:
    long long maxKelements(const vector<int>& nums, int k) {
        vector<int> exps[19];
        for (int i : nums)
            exps[18 - (int)(log(i) / log(3))].push_back(i);
            //将每个数按log(3)num为底 放在不同的桶里
            //表示可以被3/多少次
            //遍历时, 优先遍历系数最大的数
        auto ans = 0ll;
        for (auto it = exps; auto& v : exps) {
            if (k < v.size()) {
                //如果k小于当前桶的元素个数
                //则从当前桶拿取前k个元素
                //直接返回ans
                ranges::nth_element(v, v.begin() + k, greater{});
                //对当前桶进行排序, 将前k个大的数放到当前桶的前k个
                return reduce(v.begin(), v.begin() + k, ans);
                //对前k个元素进行累加
            }
            //k大于当前桶的元素个数
            ans += reduce(v.begin(), v.end(), 0ll);
            //ans对当前桶的所有元素累加
            k -= v.size();
            //k减去当前桶的元素个数
            if (it++ != exps + 18) ranges::transform(v, back_inserter(*it), [](int e) { return (e + 2) / 3; });
            //如果当前桶不是最后一个桶, 将当前桶的元素个数ceil(/3)后放入下一个桶
            //it++后此时it为下一个桶
        }
        //为什么v可以变成下一个数组?
        //auto& v : exps是范围for语法, auto it = exps仅仅是定义一个变量
        //而v会随着迭代变成下一个数组
        return ans + k;
    }
};
```

```cpp
class Solution {
public:
    long long maxKelements(vector<int>& nums, int k) {
        priority_queue<int>heap;
        for(int i = 0; i<nums.size();i++){
            heap.push(nums[i]);
        }
        // /实现大根堆
        long long sum = 0;
        for(int i = 0; i<k;i++){
            int top = heap.top();
            heap.pop();
            sum += top;
            heap.push(ceil(top/3.0));
        }  
        return sum;
    }
};
//大根堆, 每次进行++
//然后push运算后的
```

## 原地堆化
```cpp
class Solution {
public:
    long long maxKelements(vector<int> &nums, int k) {
        make_heap(nums.begin(), nums.end()); // 原地堆化（最大堆）
        long long ans = 0;
        while (k--) {
            pop_heap(nums.begin(), nums.end()); // 把堆顶移到末尾
            ans += nums.back();
            nums.back() = (nums.back() + 2) / 3;
            push_heap(nums.begin(), nums.end()); // 把末尾元素入堆
        }
        return ans;
    }
};
```
