# 滑动窗口
```c
#define MAX(a, b) ((b) > (a) ? (b) : (a))
double findMaxAverage(int* nums, int numsSize, int k) {
    int ans=INT_MIN;
    //val可能是负数,为确保ans初始小于val
    //把ans设为int类型的最小值
    int val =0;
    for(int i =0;i<numsSize;i++){
        val+=nums[i];
        if(i-k+1 < 0){
            continue;
        }
        ans =MAX(ans,val);
        val-=nums[i-k+1];
    }
    return ((double)ans)/k;
}
```
