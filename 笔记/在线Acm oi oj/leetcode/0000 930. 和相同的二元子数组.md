正难则反
算出>=goal 和>goal的个数 两者相减 就得出=goal的个数
类似滑动窗口
```cpp
class Solution {
public:
    int numSubarraysWithSum(vector<int>& nums, int goal) {
        int left1 = 0;
        int left2 = 0;
        int right = 0;
        int n = nums.size();
        int sum1 = 0;
        int sum2 = 0;
        int ans = 0;
        while(right < n){
            sum1 += nums[right];
            // 计算大于goal 的个数
            while(left1 <= right && sum1 > goal){
                sum1 -= nums[left1++];
            }
            sum2 += nums[right];
            // 计算大于等于goal的个数
            while(left2 <= right && sum2 >= goal){
                sum2 -= nums[left2++];
            }
            // 正难则反（算出>=goal 和>goal的个数 两者相减 不就得出=goal的个数么？）
            ans += left2 - left1;
            right++;
        }  
        return ans;
    }
};
```
