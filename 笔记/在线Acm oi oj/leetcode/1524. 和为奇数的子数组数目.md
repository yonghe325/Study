哈希表统计前缀和
```cpp
class Solution {
public:
    int numOfSubarrays(vector<int>& arr) {
        vector<int>hash(2,0);
        long long sum = 0;
        long long ret = 0;
        for(int i =0; i<arr.size();i++){
            // if(arr[i]%2) ret++;
            sum+=arr[i];
  
            // ret+=sum%2;
            // 等价于 if(sum%2) ret++;
            //当前前缀和为奇数, ret++
            ret+=hash[(sum+1)%2] + sum%2;//整合前面的ret+=sum%2;
            //寻找前面的奇数, 偶数前缀和
            hash[sum%2]++;
            //计算完, 将当前前缀和放入hash, 防止重复计算
        }
        return ret % 1000000007;
    }
};
```
