class Solution {
public:
    int minLength(vector<int>& nums, int k) {
        unordered_map<int, int> count;  // 统计窗口内每个元素的出现次数
        int sum = 0;                     // 不同元素的和
        int left = 0;
        int ret = INT_MAX;
        for(int right = 0; right < nums.size(); right++) {
            // 如果是新元素，加入sum
            if(count[nums[right]] == 0) {
                sum += nums[right];
            }
            count[nums[right]]++;
            // 当不同元素的和 >= k 时，尝试缩小窗口
            while(sum >= k && left <= right) {
                // 更新最小长度
                ret = min(ret, right - left + 1);
                // 移动左指针
                count[nums[left]]--;
                // 如果该元素在窗口中完全消失，从sum中减去
                if(count[nums[left]] == 0) {
                    sum -= nums[left];
                }
                left++;
            }
        }
        return ret == INT_MAX ? -1 : ret;
    }
