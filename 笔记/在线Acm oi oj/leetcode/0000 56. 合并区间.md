# 排序 + 合并左端点
```cpp
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        ranges::sort(intervals); // 按照左端点从小到大排序
        vector<vector<int>> ans;
        for (auto& p : intervals) {
            if (!ans.empty() && p[0] <= ans.back()[1]) { // 可以合并
                ans.back()[1] = max(ans.back()[1], p[1]); // 更新右端点最大值
            } else { // 不相交，无法合并
                ans.emplace_back(p); // 新的合并区间
            }
        }
        return ans;
    }
};
```
# 加系数 + 差分数组
```cpp
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        //a[i] : intervals 中元素覆盖次数
        //[start,end] 全++
        //求 a 中连续 非0 的区间
        //[1,4] [5,6] 区间的结果会不对，所以给a 扩大2倍
        //换句话说，让区间[start * 2,end * 2] 全 ++
        //这样就变成了[2,8] [10,12]
        //就可以把[1,4] [5,6] 区间分成两个不相邻的区间
        //当最后运算时, 在/2
        int mx = ranges::max(intervals,{},[](const auto& a){return a[1];})[1];
        vector<int> d(2 * mx + 2,0);
        for(auto& x : intervals){
            int start = x[0],end = x[1];
            d[start * 2] ++;
            d[end * 2 + 1] --;
        }
  
        partial_sum(d.begin(),d.end(),d.begin());
  
        //求a 中的连续 非0 的区间
        //这是典型的分组循环问题
        int i = 0,n = 2 * mx + 2;
        vector<vector<int>> ans;
        while(i < n){
            //跳过不满足的组，确定满足的组的左边界
            if(d[i] == 0){
                i ++;
                continue;
            }
            int left = i;
            while(i < n && d[i] != 0)
                i ++;
            //左闭右开区间
            ans.push_back({left / 2,(i - 1) / 2});
        }
        return ans;
    }
};
```
