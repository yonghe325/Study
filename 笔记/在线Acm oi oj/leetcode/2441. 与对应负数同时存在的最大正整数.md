class Solution {
public:
    int findMaxK(vector<int>& nums) {
        int ret = -1;
        map<int,int>arr;
        for(int i = 0; i<nums.size();i++){
            arr[nums[i]]++;
            if(arr.count(-(nums[i]))){
                ret = max(ret,abs(nums[i]));
            }
        }
        return ret;
    }
};
