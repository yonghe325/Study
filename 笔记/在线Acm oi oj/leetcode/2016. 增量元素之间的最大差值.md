class Solution {
public:
    int maximumDifference(vector<int>& nums) {
        int min_index = 0;
        int ret = -1;
        for(int i = 0; i<nums.size();i++){
            if(nums[i]<nums[min_index]){
                min_index = i;
            }
            else{
                ret = max(nums[i]-nums[min_index],ret);
            }
        }
        return ret?ret:-1;
    }
};

class Solution { 
public: 
	int maximumDifference(vector<int>& nums) { 
	int ans = 0; 
	int pre_min = INT_MAX; 
	for (int x : nums) { 
		ans = max(ans, x - pre_min); // 把 x 当作 nums[j] 
		pre_min = min(pre_min, x); // 把 x 当作 nums[i] 
	} 
	return ans ? ans : -1; } };
