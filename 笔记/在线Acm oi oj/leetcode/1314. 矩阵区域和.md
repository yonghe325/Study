class Solution {
public:
    vector<vector<int>> matrixBlockSum(vector<vector<int>>& mat, int k) {
        int row = mat.size();
        int col = mat[0].size();
        vector<vector<int>>sum(row+1,vector<int>(col+1));
        for(int i =0; i<row;i++){
            for(int j = 0;j<col;j++){
                sum[i+1][j+1] = sum[i+1][j] + sum[i][j+1] - sum[i][j] + mat[i][j];
            }
        }
        //统计所有的矩阵和
        vector<vector<int>>answer(row,vector<int>(col));
        for(int i = 0;i<row;i++){
            for(int j =0 ;j<col;j++){
                int r1 = i-k < 0? 0: i-k;
                int c1 = j-k < 0? 0: j-k;
                //防止越界
                int r2 = i+k > row-1? row-1:i+k;
                int c2 = j+k > col-1? col-1:j+k;
                //防止越界
                cout<<r1<<" "<<c1<<" "<<r2<<" "<<c2<<endl;
                answer[i][j] = sum[r2+1][c2+1] - sum[r2+1][c1] - sum[r1][c2+1] + sum[r1][c1];
            }
        }
        return answer;
    }
};
//即返回[i][j]以k为半径的矩阵和
//[i+k][j+k]为右下角坐标
//[i-k][j-k]为左上角坐标
