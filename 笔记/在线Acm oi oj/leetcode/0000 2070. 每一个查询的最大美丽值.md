```cpp deepseek优化
class Solution {
public:
    vector<int> maximumBeauty(vector<vector<int>>& items, vector<int>& queries) {
        // 1. 按价格升序排序（使用const引用避免拷贝）
        sort(items.begin(), items.end(), [](const vector<int>& a, const vector<int>& b) {
            return a[0] < b[0];
        });
        
        // 2. 预处理：每个价格位置记录到当前的最大美丽值
        for (int i = 1; i < items.size(); i++) {
            items[i][1] = max(items[i][1], items[i-1][1]);
        }
        
        // 3. 对查询下标排序
        vector<int> idx(queries.size());
        iota(idx.begin(), idx.end(), 0);
        sort(idx.begin(), idx.end(), [&](int a, int b) {
            return queries[a] < queries[b];
        });
        
        // 4. 双指针处理查询
        vector<int> ret(queries.size());
        int j = 0;
        for (int i : idx) {
            while (j < items.size() && items[j][0] <= queries[i]) {
                j++;
            }
            ret[i] = (j > 0) ? items[j-1][1] : 0;
        }
        
        return ret;
    }
};

```

# 在线运算+ 二分查找
## 前缀最大值
1. 先将item根据价格排序
2. 将美丽值更新为到当前价格为止的最大美丽值
3. 二分查找满足价格要求的位置
```cpp
class Solution {
public:
    vector<int> maximumBeauty(vector<vector<int>>& items, vector<int>& queries) {
        sort(items.begin(),items.end(),[](const vector<int>a,const vector<int>b){
            return a[0]<b[0];
        });
        //将items按照价格排序
        int max_price = 0;
        for(auto &i:items){
            max_price = max(i[1],max_price);
            i[1] = max_price;
        }
        //完成前缀最大值
        for(int& i:queries ){
            int L= 0,R = items.size();
            int mid = 0;
            while(L < R){
                mid = (L+(R-L)/2);
                if(items[mid][0] <= i){
                //查找价格相同的最后一个
                    L = mid+1;
                }
                else{
                    R = mid;
                }
            }
            //二分查找符合的美丽值
            i = (L>0)?items[L-1][1]:0;
            //L-1是满足符合要求的数
        }
        return queries;
    }
};
```
## 前缀最大值 + 过滤元素(双指针)
+ 对于相同元素, 只需要保留美丽值最大的
使用k记录之前最大的美丽值位置的下标
如果之后有美丽值小的,==直接覆盖美丽值小的物品==
只保留美丽值大的
此时有用的数组只有[0,k]
```cpp
class Solution {
public:
    vector<int> maximumBeauty(vector<vector<int>>& items, vector<int>& queries) {
        sort(items.begin(),items.end(),[](const vector<int>a,const vector<int>b){
            return a[0]<b[0];
        });
        //将items按照价格排序
        int k = 0;
        for (int i = 1; i < items.size(); i++) {
            if (items[i][1] > items[k][1]) { // 有用
                items[++k] = items[i];
            }
        }
        //去除无用物品（价格低但美丽值也低的）
        for(int& i:queries ){
            int L= 0,R = k+1;//
            int mid = 0;
            while(L < R){
                mid = (L+(R-L)/2);
                if(items[mid][0] <= i){
                    L = mid+1;
                }
                else{
                    R = mid;
                }
            }
            //二分查找符合的美丽值
            i = (L>0)?items[L-1][1]:0;
        }
        return queries;
    }
};
```
