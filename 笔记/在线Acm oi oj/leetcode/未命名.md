class Solution {
public:
    int f(string s){
        map<char,int> mid;
        for(int i =0;i<s.size();i++){
            mid[s[i]]++;
        }
        return mid.begin()->second;
    }
    vector<int> numSmallerByFrequency(vector<string>& queries, vector<string>& words) {
        //对queries和words的每个单词求出f(s)
        //对words排序
        vector<int> arr;
        for(int i =0; i<words.size();i++){
            arr.push_back(f(words[i]));
        }
        //统计最小字母出现次数
        sort(arr.begin(),arr.end());
        //排序
        vector<int> ret;
        for(int i =0;i<queries.size();i++){
            int L = 0,R = arr.size()-1;
            int mid = 0;
            int count = f(queries[i]);
            //找最后一个<=count的数
            while(L<=R){
                mid = L +(R-L)/2;
                if(arr[mid]<=count){
                    L = mid+1;
                }
                else{
                    R = mid-1;
                }
            }
            ret.push_back(arr.size()-R-1);
        }
        return ret;
    }
};
