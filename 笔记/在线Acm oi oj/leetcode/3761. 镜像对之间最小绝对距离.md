class Solution {
public:
    int reverse(int num){
        int ret = 0;
        while(num){
            ret =  (ret*10) + (num%10);
            num/=10;
        }        
        return ret;
    }
    int minMirrorPairDistance(vector<int>& nums) {
        if(nums.size() == 1) return -1;
        unordered_map<long long,int>arr;
        //以最大值作为数组长度
        // 从后向前遍历
        int n = nums.size();
        long long mid = 0;
        int ret = 100000;
        for(int i = n-1;i>=0;i--){
            mid = reverse(nums[i]);
            if(arr[mid]>0){
                ret = min(abs(arr[mid]-i),ret);
            }
            arr[nums[i]] = i;
        }
        return ret==100000?-1:ret;
    }
};

class Solution {

public:

    int minMirrorPairDistance(vector<int>& nums) {

        unordered_map<int, int> last_index;

        int n = nums.size(), ans = n;

  

        for (int j = 0; j < n; j++) {

            int x = nums[j];

            auto it = last_index.find(x);

            if (it != last_index.end()) {

                ans = min(ans, j - it->second);

            }

  

            // 计算 reverse(x)，不用字符串

            int rev = 0;

            for (; x > 0; x /= 10) {

                rev = rev * 10 + x % 10;

            }

            last_index[rev] = j;

        }

  

        return ans < n ? ans : -1;

    }

};
