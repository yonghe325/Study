class Solution {
public:
    int countCompleteSubarrays(vector<int>& nums) {
        set<int> arr;
        map<int,int> mid;
        for(int i:nums){
            arr.insert(i);
        }
        //算出种类
        int left = 0;
        int ret = 0;
        for(int i =0;i<nums.size();i++){
            mid[nums[i]]++;
            while(mid.size() == arr.size()){
                mid[nums[left]]--;
                if(!mid[nums[left]]){
                    mid.erase(nums[left]);
                }
                left++;
            }
            ret+=left;
        }
        return ret;
    }
};

class Solution {
public:
    int countCompleteSubarrays(vector<int>& nums) {
        unordered_set<int> st(nums.begin(), nums.end());
        //1. 使用无序集合减小时间复杂度
        //2. 在定义集合时, 直接使用nums的迭代器进行初始化, 减小时间复杂度
        int k = st.size();
        unordered_map<int, int> cnt;
        //不需要判断大小, 使用无序map
        int ans = 0, left = 0;
        for (int x : nums) {
            cnt[x]++;
            while (cnt.size() == k) {
                int out = nums[left];
                if (--cnt[out] == 0) {
                    cnt.erase(out);
                }
                left++;
            }
            ans += left;
        }
        return ans;
    }
};
