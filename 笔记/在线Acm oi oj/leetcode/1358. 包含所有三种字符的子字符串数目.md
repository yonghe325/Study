class Solution {
public:
    int numberOfSubstrings(string s) {
        int ret = 0;
        int right = s.size() - 1;
        map<char, int> arr;
        for (int i = s.size() - 1; i >= 0; i--) {
            arr[s[i]]++;
            while(arr.size() == 3 && i<=right) {
                ret += i + 1;
                cout<<i<<" "<<right<<endl;
                arr[s[right]]--;
                if(arr[s[right]] == 0){
                    arr.erase(s[right]);
                }
                right--;
            }
        }
        return ret;
    }
};

```cpp
class Solution {
public:
    int numberOfSubstrings(string s) {
        int ans = 0, left = 0;
        int cnt[3]{};
        for (char c : s) {
            cnt[c - 'a']++;
            while (cnt[0] && cnt[1] && cnt[2]) {
                cnt[s[left] - 'a']--;
                left++;
            }
            ans += left;
        }
        return ans;
    }
};
```
==枚举不合法的子串==
如果[left,right]不合法, 则前一个[left-1,right]一定合法
所以 从0到left 到 right 子串的个数即为符合的子串
