## 哈希表
```cpp
#define MAX(a,b) (a>b)?(a):(b);
#include <map>
class Solution {
public:
    long long maxSum(vector<int>& nums, int m, int k) {
        map<int,int> arr;
        long long sum = 0;
        int count = 0;
        long long ret = 0;
        for (int i = 0; i < nums.size(); i++) {
            arr[nums[i]]++;
            //判断是否为新元素
  
            sum+=nums[i];
            //计算最大和
            if(i-k+1<0) continue;
            //滑动窗口长度

            if(arr.size() >=m) ret = MAX(ret,sum);
            //如果符合>=m, 将sum与ret判断
            sum-=nums[i-k+1];
            if(--arr[nums[i-k+1]] == 0){
                    arr.erase(nums[i-k+1]);
            }
        }
        return ret;
    }
};
```
