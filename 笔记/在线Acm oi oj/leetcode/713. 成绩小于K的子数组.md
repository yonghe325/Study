class Solution {
public:
    int numSubarrayProductLessThanK(vector<int>& nums, int k) {
        int ret = 0;
        int left = 0;
        int mul = 1;
        for(int i =0; i<nums.size();i++){
            // if(nums[i]<=k) ret++;
            mul*=nums[i];
            while(left <= i && mul>=k ){
                mul/=nums[left];
                left++;
            }
            ret += i-left+1;
            //求出个数
            //10 | 10,5 5|
            //1  | 2     |
        }
        return ret;
    }
};
