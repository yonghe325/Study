class Solution {
public:
    int maxDigit(int num){
        int ret = 0;
        while(num){
            ret = max(ret,num%10);
            num/=10;
        }
        return ret;
    }
    int maxSum(vector<int>& nums) {
        int ret = -1;
        vector<int>arr(9,-1);
        int max_digit = 0;
        for(int& i:nums){
            max_digit = maxDigit(i)-1;
            if(arr[max_digit]>0){
                ret = max(ret,i+arr[max_digit]);
                arr[max_digit] = max(arr[max_digit],i);
            }
            else{
                arr[max_digit] = i;
            }
        }
        return ret;
    }
};
