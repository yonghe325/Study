class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int min_index = 0, max_index = 0;
        int ret = 0;
        for(int i = 0; i<prices.size();i++){
            if(prices[i]<prices[min_index]){
                min_index = i;
            }
            if(prices[i] > prices[max_index]){
                max_index = i;
                ret = max(ret,prices[max_index]-prices[min_index]);
            }
            max_index = max_index>min_index?max_index:min_index;
            //保证max始终在min右边
        }
        return ret;
    }
};

优化:
不在需要记录最大值, 当[i]大于min时, 更新差值
```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int min_index = 0;
        int ret = 0;
        for(int i = 0; i<prices.size();i++){
            if(prices[i]<prices[min_index]){
                min_index = i;
            }
            else{
                ret = max(prices[i]-prices[min_index],ret);
            }
        }
        return ret;
    }
};
```
