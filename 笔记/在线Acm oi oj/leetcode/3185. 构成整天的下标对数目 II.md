class Solution {
public:
    long long countCompleteDayPairs(vector<int>& hours) {
        vector<int>arr(24,0);
        long long ret = 0;
        int lack = 0;
        for(int& i:hours){
            lack = (24-i%24)%24;
            if(arr[lack]>0) ret+=arr[lack];
            arr[i%24]++;
        }
        return ret;
    }
};
