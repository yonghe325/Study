class Solution {
public:
    int minimumCardPickup(vector<int>& cards) {
        int ret = cards.size()+1;
        vector<int>arr(ranges::max(cards)+1,-1);
        for(int i =0; i<cards.size();i++){
            if(arr[cards[i]] > -1){
                ret = min(ret,(i - arr[cards[i]]+1));
            }
            arr[cards[i]] = i;
        }
        return ret ==cards.size()+1?-1:ret;
    }
};
