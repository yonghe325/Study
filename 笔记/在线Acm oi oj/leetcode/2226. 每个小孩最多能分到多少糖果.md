下界L最小值为0, 即所有糖果加起来< k
上界最大值为candies.max, 即最大的一堆糖果给一个小孩

```cpp
class Solution {
public:
    int maximumCandies(vector<int>& candies, long long k) {
        sort(candies.begin(),candies.end());
        int n = candies.size();
        auto check = [&](int m)->bool{
            int l = 1, r = 0;
            //这里判断m是否小于candies.min
            //如果小于, 说明k>n
            //要对每一堆糖果进行拆分, 不用再进行二分查找
            //即仅当m大于candies的最小值时查找
            if(m > candies[0]){
                l =0,r = n;
                int mm =0;
                while(l<r){
                    mm = l+(r-l)/2;
                    if(candies[mm]>=m)
                        r = mm;
                    else
                        l = mm+1;
                }
            }
            //l-1 <= m
            long long  ret = 0;
            //ret为以当前数目m,可以分多少个小孩
            //从l-1开始向后遍历, 看看以当前m为数目, 能分成多少堆糖果()
            for(int i =l-1; i<n;i++){
                ret+=candies[i]/m;
                if(ret>=k) return true;
                //m还能变大
                //L->
            }
            return false;
            //m要变小
            //<-R
        };
        int L = 0, R = candies[n-1]+1;
        int mid = 0;
        while(L+1<R){
            mid = L +(R-L)/2;
            (check(mid)?L:R) = mid;
        }  
        return L;
    }
};
```
