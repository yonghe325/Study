class Solution {
public:
    int numEquivDominoPairs(vector<vector<int>>& dominoes) {
        vector<vector<int>> arr(9, vector<int>(9,0));
        int index = 0;
        int mx = 0;
        int ret = 0;
        for(int i = 0;i<dominoes.size();i++){
            index = (dominoes[i][0]<dominoes[i][1]?dominoes[i][0]:dominoes[i][1]) -1;
            mx = dominoes[i][0] + dominoes[i][1] -index-2;
            arr[index][mx]++;
            ret += arr[index][mx]-1;
        }
        return ret;
    }
};
//即两个数组的元素相同,可以顺序不同, 视为相等
//以dominoes[i]中较小的元素为索引, 将较大的数存储
//向后遍历, 查找较小的数, 看看较大的数有几个(x), 结果加上x-1
