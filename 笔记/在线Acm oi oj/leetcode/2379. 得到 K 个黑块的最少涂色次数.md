# 滑动窗口
## if分支法
```cpp
#define MIN(a,b) ((a<b)?(a):(b))
int minimumRecolors(char* blocks, int k) {
    int mid =0;
    int ret =k+1;
    for(int i =0;blocks[i];i++){
        if(blocks[i] == 'W'){
            mid++;
        }
        if(i-k+1 < 0){
            continue;
        }
        ret =MIN(ret,mid);
        if(blocks[i-k+1] == 'W'){
            mid--;
        }
    }
    return ret;
}
```
复习:
```cpp
#define MIN(a, b) ((b) < (a) ? (b) : (a))
class Solution {
public:
    int minimumRecolors(string blocks, int k) {
        //窗口向前滑动
        //遇到"W"就将涂黑次数++
        //当"W"被移出时, 将涂黑次数--
        int count = 0;
        //涂黑次数
        //一定大于等于0
        int ret = INT_MAX;
        //返回值,最大值
        for(int i =0; blocks[i];i++){
            if(blocks[i] == 'W') count++;
            if(i-k+1 < 0) continue;
            ret = MIN(ret, count);
            if(blocks[i-k+1] == 'W') count--;
        }
        return ret;
    }
};
```
# 位运算法
由于在ASCLL码中
W= 01010111
B = 01000010
所以可以使用&判断是否为W
```c
int minimumRecolors(char* blocks, int k) {
    int mid =0;
    for(int i= 0;i<k;i++){
        mid+=blocks[i] & 1;
    }
    //求出前k个元素中"W"的个数
    int ret =mid;
    for(int i = k;i<strlen(blocks);i++){
        mid += ((blocks[i] & 1) -( blocks[i-k] &1));
		//(blocks[i] & 1)为进入滑动窗口
		//(blocks[i] & 1) == true, 说明进入一个"W"
		//( blocks[i-k] &1)为移出滑动窗口
		//( blocks[i-k] &1) == true, 说明移出一个"W"
		//只计算"W"的个数, ,且直接从k下标处开始
	    //进入"W", mid++
	    //移出"W", mid--
        ret = MIN(ret,mid);
    }
    return ret;
}
```
