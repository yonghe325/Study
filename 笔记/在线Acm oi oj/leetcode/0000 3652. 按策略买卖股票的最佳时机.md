# 错误答案
```cpp
class Solution {
public:
    long long maxProfit(vector<int>& prices, vector<int>& strategy, int k) {
        int mn_index = 0;
        int pre_mn = strategy[0];
        int sum = 0;
        int ret = 0;
        for(int i =0; i<strategy.size();i++){
            ret += strategy[i] * prices[i];
            //顺便统计原始的利润
            sum+=strategy[i];
            if(sum<=pre_mn){
                mn_index = i;
                //记录最小前缀和出现的位置
                pre_mn = sum;
            }
        }
        //此时mn_index为最小前缀和最后出现的位置
        int row_ret = ret;
        // cout<<"row_ret : "<<row_ret<<endl<<endl;
        int start_index = mn_index+1-k>=0?mn_index-k+1:0;
        //确定起始位置
        int current_index = 0;
        for(int i = 0 ;i<k;i++){
            current_index = start_index+i;
            //当前要操作的位置
            ret -= strategy[current_index]*prices[current_index];
            //先对当前位置进行-=;
            //如果该位置由-1->0, 不进行if
            //0->0, 无变化
            //1-0,-
            //0->1进行if
            //1->1, 进行if
            //-1->1, 进行if
            if(i>=k/2){
                ret +=prices[current_index];
            }
        }
        return ret>row_ret?ret:row_ret;
    }
};
// I
//  先统计按原始策略获得的利润
//  以k为大小滑动遍历strategy,计算得到的最大利润
// II
//  直接遍历strategy, -1越少越好
//  统计strategy前缀和, 把k顶头放到在前缀和最小处最后出现的地方
//   若所有数都大于0
//
```
无法判断-1处是否过大
# 灵神
+ 定义数组 c，其中 c[i]=prices[i]⋅strategy[i]。计算 c 的前缀和，记作 sum。
+ 计算 prices 的前缀和，记作 sumSell。
+ 如果不修改，答案为 sum[n]。
+ 如果修改，枚举修改子数组 [i−k,i−1]。修改后的利润由三部分组成：
	$[0,i−k−1] 的 prices[i]⋅strategy[i] 之和，即 sum[i−k]$
		左边部分的总和
	$[i,n−1] 的 prices[i]⋅strategy[i] 之和，即 sum[n]−sum[i]$
		右边部分的总和
	$[i−k/2,i−1] 的 prices[i] 之和，即 sumSell[i]−sumSell[i−k/2]$
		中间部分 要改成1的部分的总和
	由于 中间部分 要改成0的部分的总和总和就为0
		所以最后不包括中间部分 要改成0的部分的总和
+ 总和为 : $$sum[i−k]+sum[n]−sum[i]+sumSell[i]−sumSell[i−k/2]$$
```cpp
class Solution {
public:
    long long maxProfit(vector<int>& prices, vector<int>& strategy, int k) {
        int n = prices.size();
        vector<long long> sum(n + 1), sum_sell(n + 1);
        for (int i = 0; i < n; i++) {
            sum[i + 1] = sum[i] + prices[i] * strategy[i];
            sum_sell[i + 1] = sum_sell[i] + prices[i];
        }
  
        long long ans = sum[n]; // 不修改
        for (int i = k; i <= n; i++) {
			//从k到n遍历
            long long res = sum[i - k] + sum[n] - sum[i] + sum_sell[i] - sum_sell[i - k / 2];
            ans = max(ans, res);
        }
        return ans;
    }
};
```
