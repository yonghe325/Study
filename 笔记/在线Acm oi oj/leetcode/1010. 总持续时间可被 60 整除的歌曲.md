class Solution {
public:
    int numPairsDivisibleBy60(vector<int>& time) {
        vector<int>arr(60,0);
        int ret = 0;
        int lack = 0;
        for(int& i: time){
            lack = (60-i%60)%60;
            //当前i缺少的
            if(arr[lack]>0) ret +=arr[lack];
            arr[i%60]++;
            // cout<<lack<<" "<<arr[lack]<<endl;
        }
        return ret;
    }
};
//以time[i]%60为索引
