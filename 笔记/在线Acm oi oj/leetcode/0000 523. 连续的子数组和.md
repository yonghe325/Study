类似于[[0000 974. 和可被 K 整除的子数组]]
# 前缀和 + 哈希表
运用 ==同余定理==
+ 同余定理 : 如果两个整数m、n满足n-m能被k整除，那么n和m对k同余
## 自己写的
```cpp
class Solution {
public:
    bool checkSubarraySum(vector<int>& nums, int k) {
        if(nums.size()<2) return false;
        map<int,int>hash;
        int sum = 0;
        for(int i =0; i<nums.size();i++){
            sum+=nums[i]%k;
            //计算余数
            if(!(sum%k) || hash[sum%k]>0 ){
                    return true;
                //当前前缀和能被k整除
                //或
                //hash中有满足条件的余数
                //即减去这个余数可以使当前数组满足好数组
                return true;
  
            }
            hash[nums[i]%k]++;
            //将当前元素的余数放入hash中
        }
        return false;
    }
};
```
+ 缺陷 : 
	+ 无法计算 0 为唯一好数组元素的情况
		例如: \[5,0,0,0,]  k = 3
		此时 下标\[1,2]满足好数组
		但本算法无法计算该情况
## 官方题解
```cpp
class Solution {
public:
    bool checkSubarraySum(vector<int>& nums, int k) {
        int m = nums.size();
        if (m < 2) {
            return false;
        }//长度小于2, 直接返回false
        unordered_map<int, int> mp;
//关键
        //<前缀和余数, 该前缀和余数最后一次出现的位置>
        mp[0] = -1;
//这是关键 , 将0的下标初始设为-1
        //保证当好数组唯一元素为0时, 可以被视为好数组
        int remainder = 0;
        //余数前缀和
        for (int i = 0; i < m; i++) {
            remainder = (remainder + nums[i]) % k;
            //统计当前的余数前缀和
            if (mp.count(remainder)) {
	            //如果有当前同余的前缀和存在于mp中
	            //进入判断长度的流程
                int prevIndex = mp[remainder];
                //prevIndex为上一次同余前缀和的下标
                if (i - prevIndex >= 2) {
	                // 长度大于2, 返回true
                    return true;
                }
            } else {
                mp[remainder] = i;
            }
        }
        return false;
    }
};
```
