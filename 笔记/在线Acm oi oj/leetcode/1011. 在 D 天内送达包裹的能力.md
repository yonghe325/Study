# 原数组模拟 + 二分答案
```cpp
class Solution {
public:
    int shipWithinDays(vector<int>& weights, int days) {
        // 检查给定运载能力 m 是否能在 days 天内运完
        auto check = [&](int m) -> bool {
            int need = 1;       // 需要的天数
            int cur = 0;         // 当前这趟已经装的重量
            for (int w : weights) {
                if (cur + w > m) {
                    // 装不下了，新开一趟
                    need++;
                    cur = w;
                } else {
                    cur += w;
                }
            }
            return need <= days;
        };
        // 下界：至少要能装下最重的一个货物
        int L = *max_element(weights.begin(), weights.end());
        // 上界：一次装完所有货物
        int R = accumulate(weights.begin(), weights.end(), 0);
  
        // 二分查找最小可行容量
        while (L < R) {
            int mid = L + (R - L) / 2;
            if (check(mid)) {
                R = mid;        // mid 可行，尝试更小
            } else {
                L = mid + 1;    // mid 不可行，必须增大
            }
        }
        return L;
    }
};
```
