class Solution {
public:
    long long countSubarrays(vector<int>& nums, long long k) {
        long long left = 0;
        long long sum = 0;
        long long ret = 0;
        for(int i =0;i<nums.size();i++){
            sum+=nums[i];
            while(sum*(i-left+1) >= k){
                sum-=nums[left];
                left++;
            }
            ret += i-left+1;
        }
        return ret;
    }
};
