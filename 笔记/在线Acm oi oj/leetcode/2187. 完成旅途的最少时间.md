class Solution {
public:
    long long minimumTime(vector<int>& time, int totalTrips) {
        auto check = [&](long long m)->bool{
            cout<<m<<endl;
            long long ret = 0;
            for(int i:time){
                ret += m/i;
                if(ret>=totalTrips) return true;
                //已经大于等于totalTrips, 左移R
            }
            // cout<<m<<" "<<ret<<endl;
            return false;
            //小于totalTrips, 右移L
        };
        sort(time.begin(),time.end());
        //先排序
        // long long L = time[0], R = time[time.size()-1]*totalTrips;
        long long L = time[0], R = (long long)time[0] * totalTrips;
        //防溢出
  
        long long mid = 0;
        while(L+1<R){
            mid = L +(R-L)/2;
            (check(mid)?R:L) = mid;
        }
        if(L==time[0]) return L;
        return R;
    }
};
