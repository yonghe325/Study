用最简分数作为键
避免精度问题
```cpp
class Solution {
public:
    long long interchangeableRectangles(vector<vector<int>>& rectangles) {
        map<pair<int,int>, int> arr;  // 用最简分数作为键
        long long ret = 0;
        for (auto& r : rectangles) {
            int w = r[0], h = r[1];
            int g = gcd(w, h);
            pair<int,int> ratio = {w / g, h / g};
            ret += arr[ratio];
            arr[ratio]++;
        }
        return ret;
    }
};
```
