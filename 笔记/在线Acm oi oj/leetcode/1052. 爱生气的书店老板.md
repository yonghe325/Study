滑动窗口+过滤
```cpp
#define MAX(a,b) (a>b)?(a):(b);
class Solution {
public:
    int maxSatisfied(vector<int>& customers, vector<int>& grumpy, int minutes) {
        //先使用grumpy对customers进行过滤, 得到新的数组
        //使用新数组进行运算
        int sum = 0;
        int mid = 0;
        int ret = 0;
        for(int i =0; i<customers.size();i++){
            //直接过滤到customers
            //一边过滤, 一边使用滑动窗口, 一边进行sum求和
            sum += (!grumpy[i] ? customers[i]:0);
            mid += ((grumpy[i]) ? customers[i]:0);
            int left = i-minutes+1;
            if(left<0) continue;
            ret = MAX(ret,mid);
            mid -= grumpy[left] & 1 ? customers[left]:0;
        }
        return sum+ret;
    }
};
```
