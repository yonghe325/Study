class SnapshotArray {
private:
    vector<int>arr;
    vector<vector<int>>snap_arr;
    int snap_id;
public:
    SnapshotArray(int length) {
        vector<int>(length, 0).swap(arr);
        vector<vector<int>>().swap(snap_arr);
        snap_id= 0;
    }
    //初始化数组
    void set(int index, int val) {
        int local_snap_id = snap_id;
        snap_arr.push_back({local_snap_id,index,val});
        //存放快照
        arr[index] = val;
        //修改原数组
    }
    int snap() {
        return snap_id++;
    }
    int get(int index, int snap_id) {
        //snap_arr是按照修改顺序填入人的, 所以后面一定覆盖前面的
        //二分查找最后满足快照编号
        //在向左遍历
        int L = 0, R = snap_arr.size();
        int mid = 0;
        while(L <R){
            mid = L + (R-L)/2;
            if(snap_arr[mid][0] <= snap_id){
                L = mid+1;
            }
            else{
                R = mid;
            }
        }
        //遍历到最右边
        for(int i  = L - 1; i>=0;i--){
            if(snap_arr[i][1] == index){
                return snap_arr[i][2];
            }
        }
        //返回该下标最后的修改痕迹
        // 没找到返回0
        return 0;
    }

};
