不定长滑动窗口
```cpp
#define MAX(a,b) a>b?a:b
#include <cmath>
class Solution {
public:
    int equalSubstring(string s, string t, int maxCost) {
        int sum = 0;
        int left = 0;
        int ret = 0;
        for(int i =0; i<s.size();i++){
            sum+=abs(s[i]-t[i]);
            while(sum>maxCost){
                sum -= abs(s[left]-t[left]);
                left++;
            }
            ret = MAX(ret,i-left+1);
        }
        return ret;
    }
};
```
