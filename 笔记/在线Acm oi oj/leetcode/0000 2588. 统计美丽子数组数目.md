等同于[[560. 和为 K 的子数组]]
相当于把该题条件 " 和为k的子数组"
换为 "异或和为0"
+ `unordered_map<int, int> cnt{{0, 1}};`的作用
	转到![[560. 和为 K 的子数组#`hash[0]++;`的作用]]
```cpp
class Solution {
public:
    long long beautifulSubarrays(vector<int>& nums) {
        long long ans = 0;
        int s = 0;
        unordered_map<int, int> cnt{{0, 1}};
        for (int x : nums) {
            s ^= x;
            ans += cnt[s]++;
        }
        return ans;
    }
};
```
