子序列: 不改变元素出现的顺序, 可以不连续
所以可以对数组进行排序
排序前后元素的顺序不影响选取时元素的实际顺序
## 排序 + 滑动窗口
```cpp
class Solution {
public:
    vector<int> answerQueries(vector<int>& nums, vector<int>& queries) {
        sort(nums.begin(),nums.end());
        vector<int> answer;
        for(int k:queries){
            int left = 0 ;
            int len = 0;
            int sum = 0;
            //1 2 4 5
            // 3 10 21
            for(int i =0; i<nums.size();i++){
                sum+=nums[i];
                //1 2 4 5
                while(sum>k){
                    sum-=nums[left];
                    left++;
                }
                // if(i == left) break;
                //当[i] == [left]时
                //说明当前nums已经没有比k小的值了, 长度即为0
                len = max(len,i-left+1);
            }
            answer.push_back(len);
        }
        return answer;
    }
};
```
## 排序 + 前缀和 + 二分算法
```cpp
class Solution {
public:
    vector<int> answerQueries(vector<int>& nums, vector<int>& queries) {
        sort(nums.begin(),nums.end());
        for(int i =1; i<nums.size();i++){
            nums[i]+=nums[i-1];
        }
        //统计前缀和
        
        for(int& i : queries){
            int L = 0, R = nums.size();
            int mid = 0;
            while(L< R){
                mid = L + ((R-L)>>1);
                if( nums[mid] <= i){//定位到第一个大于i的位置
                    L = mid+1;
                }
                else{
                    R = mid;
                }
            }
			//此时len最大值即为当前L的下标
            i = L;    

        }
        //统计子序列最大值
        return queries;
    }
};
```
