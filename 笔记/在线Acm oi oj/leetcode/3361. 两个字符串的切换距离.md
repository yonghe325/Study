# 顺逆时针前缀和
```cpp
class Solution {
public:
    long long shiftDistance(string s, string t, vector<int>& nextCost, vector<int>& previousCost) {
        int n = 26;
        vector<long long> next_sum(n + 1), pre_sum(n + 1);
        for (int i = 0; i < n; i++) {
            next_sum[i + 1] = next_sum[i] + nextCost[i];
            pre_sum[i + 1] = pre_sum[i] + previousCost[i];
        }
        long long ret = 0;
        for (int i = 0; i < s.size(); i++) {
            int from = s[i] - 'a';
            int to = t[i] - 'a';
            // 顺时针
            long long next;
            if (from <= to) {
                next = next_sum[to] - next_sum[from];
            } else {
                next = (next_sum[n] - next_sum[from]) + next_sum[to];
            }
            // 逆时针
            long long pre;
            if (from >= to) {
                pre = pre_sum[from + 1] - pre_sum[to + 1];
            } else {
                pre = pre_sum[from + 1] + (pre_sum[n] - pre_sum[to + 1]);
            }
            ret += min(next, pre);
        }
        return ret;
    }
};
```

# 灵神 : 展开环行前缀和
多统计一次字母回到本身的前缀和
即每个字母绕回本身的26个长度的前缀和
即一共52个长度, 多一个前导0
共53个前缀的前缀和数组
```cpp
class Solution {
public:
    long long shiftDistance(string s, string t, vector<int>& nextCost, vector<int>& previousCost) {
        const int SIGMA = 26;
        long long nxt_sum[SIGMA * 2 + 1]{}, pre_sum[SIGMA * 2 + 1]{};
        for (int i = 0; i < SIGMA * 2; i++) {
            nxt_sum[i + 1] = nxt_sum[i] + nextCost[i % SIGMA];
            pre_sum[i + 1] = pre_sum[i] + previousCost[i % SIGMA];
        }
  
        long long ans = 0;
        for (int i = 0; i < s.size(); i++) {
            int x = s[i] - 'a', y = t[i] - 'a';
            ans += min(nxt_sum[y < x ? y + SIGMA : y] - nxt_sum[x],
                       pre_sum[(x < y ? x + SIGMA : x) + 1] - pre_sum[y + 1]);
        }
        return ans;
    }
};
```
