class Solution {
public:
    int minimumSum(vector<int>& nums) {
        int n = nums.size();
        if(n<3) return -1;
        vector<int>right_min(n);
        int Rmin = nums[n-1];
        for(int i = n-2;i>0;i--){
            right_min[i] = Rmin;
            Rmin = min(nums[i],Rmin);
        }
        //计算该下标右边的最小值(不包括该下标)
  
        int ret = INT_MAX;
        int Lmin = nums[0];
        for(int i =1; i<n-1;i++){
            if(nums[i]>Lmin && nums[i]>right_min[i]) ret = min(nums[i]+Lmin+right_min[i],ret);
            Lmin = min(Lmin,nums[i]);
            // nums[i] =
            // cout<<i<<" "<<ret<<" "<< Lmin<< " "<<right_min[i]<<endl;
        }
        return ret==INT_MAX?-1:ret;
    }
};
