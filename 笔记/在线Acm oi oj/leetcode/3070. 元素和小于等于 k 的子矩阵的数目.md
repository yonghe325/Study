# 二维前缀和
## self
```cpp
class Solution {
public:
    int countSubmatrices(vector<vector<int>>& grid, int k) {
        int row = grid.size(), col = grid[0].size();
        vector<vector<int>>sum(row+1,vector<int>(col+1));
        int ret =0;
        for(int i = 0; i<row;i++){
            for(int j =0;j<col;j++){
                sum[i+1][j+1] = sum[i+1][j]  + sum[i][j+1] - sum[i][j] + grid[i][j];
                if(sum[i+1][j+1] <= k){
                    ret++;
                }
            }
        }
        return ret;
    }
};
//矩阵求前缀和, 同时判断该矩阵和是否<=k
```

```cpp
            for (int j = 0; j < n; j++) {

                ans += sum[i + 1][j + 1] <= k;
                //省去if判断
            }
        }
        return ans;
    }
```

# 维护每列的元素和

```cpp
class Solution {
public:
    int countSubmatrices(vector<vector<int>>& grid, int k) {
        int n = grid[0].size();
        vector<int> col_sum(n);
        int ans = 0;
        for (auto& row : grid) {
            int s = 0;
            for (int j = 0; j < n; j++) {
                col_sum[j] += row[j];
                s += col_sum[j];
                if (s > k) {
                    break;
                }
                ans++;
            }
        }
        return ans;
    }
};
```
