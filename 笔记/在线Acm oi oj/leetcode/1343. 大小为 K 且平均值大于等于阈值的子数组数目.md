# 滑动窗口
```c
int numOfSubarrays(int* arr, int arrSize, int k, int threshold) {
    int count =0;
    int sum =0;
    for(int i =0;i<arrSize;i++){
        sum+=arr[i];
        if(i-k+1<0){
            continue;
        }
        if(((double)sum)/k >= threshold){
            count++;
        }
        sum-=arr[i-k+1];
    }
    return count;
}
```
