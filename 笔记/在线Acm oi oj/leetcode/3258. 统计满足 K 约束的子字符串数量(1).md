class Solution {
public:
    int countKConstraintSubstrings(string s, int k) {
        int arr[2] = {0};
        int left = 0;
        int ret = 0;
        for(int i=0;i<s.size();i++){
            arr[(s[i]-48)]++;
            while(arr[0]>k && arr[1]>k){
                arr[s[left]-48]--;
                left++;
            }
            cout<<left<<"  "<<i<<endl;
            ret += i-left+1;
            //直接返回区间内的个数
        }
        return ret;
    }
};
