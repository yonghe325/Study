不定长滑动窗口
```cpp
#define MAX(a,b) a>b?a:b
class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        unordered_map<int,int> arr;
        //求种类恰好为2
        //且数量最多的范围
        int ret = 0;
        int left = 0;
        int sum = 0;
        for(int i = 0; i<fruits.size();i++){
            arr[fruits[i]]++;
            while(arr.size()>2){
                arr[fruits[left]]--;
                if(!arr[fruits[left]]) arr.erase(fruits[left]);
                left++;
            }
            sum = i-left+1;
            ret = MAX(ret,sum);
        }
        return ret;
    }
};
```
