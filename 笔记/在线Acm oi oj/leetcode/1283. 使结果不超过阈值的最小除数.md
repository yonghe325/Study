class Solution {
public:
    int Upsum(vector<int>& nums,int M){
        cout<<M<<endl;
        int L = 0, R = nums.size();
        int mid = 0,ret = 0;
        while(L <R){
            mid = L + (R-L)/2;
            if(nums[mid]<M){
                L = mid+1;
            }
            else{
                R = mid;
            }
        }
        //得到第一个大于等于M的下标
        ret += L;
        cout<<ret<<endl;
        //如果nums[mid] 恰好等于M, ret+=mid+1
        //否则ret+=mid
        for(int i = L;i<nums.size();i++){
            ret += ceil((double)nums[i]/M);
            // (nums[i] + M - 1) / M;
        }
        cout<<ret<<endl<<endl;
        return ret;
    }
    int smallestDivisor(vector<int>& nums, int threshold) {
        sort(nums.begin(),nums.end());
        //先进行排序
        int L = 1, R = nums[nums.size()-1];
        int mid =0 ;
        while(L < R){
            mid = L +(R-L)/2;
            if(Upsum(nums,mid) <= threshold){
                R = mid;
            }
            else{
                L = mid+1;
            }
        }
        return R;
    }
};
