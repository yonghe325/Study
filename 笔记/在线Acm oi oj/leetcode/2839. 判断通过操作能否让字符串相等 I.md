class Solution {
public:
    bool canBeEqual(string s1, string s2) {
        for(int i =2; i<s1.size();i++){
            if(s1[i] == s2[i]){
                continue;
            }
            if(s1[i] != s2[i] && (s1[i] == s2[i-2] && s2[i] == s1[i-2])){
                s2[i] = s1[i];
                s2[i-2] = s1[i-2];
            }
            else{
                return false;
            }
        }
        if(s1 == s2){
            return true;
        }
        else{
            return false;
        }
    }
};

class Solution {
public:
    bool canBeEqual(string s1, string s2) {
        int cnt1[2][26]{}, cnt2[2][26]{};
        for (int i = 0; i < s1.length(); i++) {
            cnt1[i % 2][s1[i] - 'a']++;
            cnt2[i % 2][s2[i] - 'a']++;
        }
        return memcmp(cnt1, cnt2, sizeof(cnt1)) == 0;
    }
};
