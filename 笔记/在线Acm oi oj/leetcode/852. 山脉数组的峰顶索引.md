int peakIndexInMountainArray(int* arr, int arrSize) {
//二分查找
    int *left = &arr[0];
    int *right = &arr[arrSize-1];
    int index = 0;
    int len  =arrSize;
    while(len){
        len/=2;
        if(  *(left + len) >*(left + len + 1) ){
//每次都从left的位置出发,index为相对left初始位置的长度
//保证当循环结束时,left为峰顶
            right  =(left + len);
        }
        else{
            left =(left + len +1);
            index += (len+1);
            if( *(left + len/2)  == arr[arrSize-1] ){
                return index;
            }
//由于山峰一定不在边界,所以当left下一个是边界时,直接跳出
        }
    }
    return index;
}
