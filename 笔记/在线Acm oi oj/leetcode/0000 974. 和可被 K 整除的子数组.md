# 哈希表 + 前缀和
## 灵神
把负数前缀和调整为非负数。
如果不调整负数，会出现同余的数据没有汇总的情况
例如 k=4，−1 和 3 关于模 k 同余，如果不调整 −1，那么哈希表会分开保存 −1 和 3
正确做法是==把 −1 的个数汇总到 3 ==上
```cpp
class Solution {
public:
    int subarraysDivByK(vector<int>& nums, int k) {
        unordered_map<int, int> cnt;
        int ans = 0, s = 0;
        for (int x : nums) {
            cnt[s]++;
            s = (s + x % k + k) % k; // 保证 s 非负
            ans += cnt[s];
        }
        return ans;
    }
};
```
## self
```cpp
class Solution {
public:
    int subarraysDivByK(vector<int>& nums, int k) {
        map<int,int>hash;
        long long sum =0 ;
        int ret  = 0;
        for(int i =0; i<nums.size();i++){
            sum += nums[i];
            // ret +=sum%k?0:1;
            //能被k整除, ret++
            ret+=(sum%k?0:1) + hash[+sum%k];
            if(sum>=0)
                ret+=hash[-(k-sum%k)];
                //寻找 +sum%k 或 -(k-sum%k);
            else
                ret+=hash[k+sum%k];
                //寻找 + (sum%k)  或 k + (sum%k)
            hash[sum%k]++;
            // cout<<i<<" "<<sum%k<<" "<<hash[sum%k]<<endl;
        }  
        return ret;
    }
};

//当计算到新的前缀和时, 寻找满足余数的元素
```
