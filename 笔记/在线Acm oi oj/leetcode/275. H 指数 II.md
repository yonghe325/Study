至少有h篇论文引用次数大于等于h
由于数组是升序的, 所以对于第h篇论文, 只需要查询倒数第h篇论文的引用数是否大于等于h
```cpp
class Solution {
public:
    int hIndex(vector<int>& citations) {
        int n = citations.size();
        int L = 0, R = n+1;
        int mid = 0;
        while(L+1<R){
            mid = L+(R-L)/2;
            if( citations[n-mid] >= mid){
            //citations[n-mid]为倒数第h篇论文的引用数
                //满足
                L= mid;
            }
            else{
                //不满足
                R = mid;
            }
        }
        return L;
    }
};
//使用二分答案模板
//求最大
```
