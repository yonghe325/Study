```cpp
class Solution {
public:
    long long countBadPairs(vector<int>& nums) {
        unordered_map<int, int>arr;
        long long ret = 0;
        for(int i = 0; i<nums.size();i++){
            arr[nums[i]-i]++;
            ret+=arr[nums[i]-i]-1;
        }
        return (nums.size() * (nums.size() - 1) / 2)-ret;
    }
};
//统计相对于下标的差值
//当两个位置的差值相同, 则不是坏数对
//4 1 3 3
//4 0 1 0
```

class Solution {
public:
    long long countBadPairs(vector<int>& nums) {
        int n = nums.size();
        long long ans = 1LL * n * (n - 1) / 2;
        unordered_map<int, int> cnt;
        for (int i = 0; i < n; i++) {
            ans -= cnt[nums[i] - i]++;
        }
        return ans;
    }
};
