```cpp
class Solution {
public:
    int maximumSum(vector<int> &nums) {
        int ans = -1;
        vector<int> mx(82, INT_MIN); // 至多 9 个 9 相加
        for (int num : nums) { // 枚举 num = nums[j]
            int s = 0; // num 的数位和
            for (int x = num; x; x /= 10) { // 枚举 num 的每个数位
                s += x % 10;
            }
            ans = max(ans, mx[s] + num); // 左边找一个数位和也为 s 的最大的 nums[i]
            mx[s] = max(mx[s], num); // 维护数位和等于 s 的最大元素
        }
        return ans;
    }
};
```
