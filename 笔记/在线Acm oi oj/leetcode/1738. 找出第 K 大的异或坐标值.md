class Solution {
public:
    int kthLargestValue(vector<vector<int>>& matrix, int k) {
        int row = matrix.size(), col = matrix[0].size();
        vector<vector<int>>sum(row+1,vector<int>(col+1));
        map<int,int>hash;
        for(int i = 0;i<row;i++){
            for(int j = 0; j<col;j++){
                sum[i+1][j+1] = sum[i+1][j] ^ sum[i][j+1] ^ sum[i][j] ^ matrix[i][j];
                hash[sum[i+1][j+1]]++;
            }
        }
        // 找第 k 大的值（map 默认升序，从后往前）
        int cnt = 0;
        for (auto it = hash.rbegin(); it != hash.rend(); it++) {
            cnt += it->second;
            if (cnt >= k) {
                return it->first;
            }
        }
        return -1;
    }
};
//求出所有异或和
//排序寻找第k个

class Solution {
public:
    int kthLargestValue(vector<vector<int>>& matrix, int k) {
        int m = matrix.size(), n = matrix[0].size();
        vector<int> a;
        vector<vector<int>> s(m + 1, vector<int>(n + 1));
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                s[i + 1][j + 1] = s[i + 1][j] ^ s[i][j + 1] ^ s[i][j] ^ matrix[i][j];
            }
            a.insert(a.end(), s[i + 1].begin() + 1, s[i + 1].end());
        }
        ranges::nth_element(a, a.end() - k);
        return a[a.size() - k];
    }
};
