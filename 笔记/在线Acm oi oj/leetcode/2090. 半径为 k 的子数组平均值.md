# 滑动窗口
## 动态填-1
```
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* getAverages(int* nums, int numsSize, int k, int* returnSize) {
    int* arr =(int*)malloc(sizeof(int)*numsSize);
    *returnSize =numsSize;
    long long sum =0;
    int mid =0;
    //使用mid当arr的计数器
    for(int i=0;i<numsSize;i++){
        sum+=nums[i];
        
        if(k==0){
	    //半径为0时,每次i移动,都清除sum
            arr[mid] = sum;
            mid++;
            sum-=nums[i];
        }
        else if( i-k+1 <=0){
        //当i小于k+1,即i小于半径时,将i之前填为-1
            arr[mid] =-1;
            mid++;
            continue;
        }
        //当i处于k+与2k+1之间时
        //mid不变,i向前走
        else if(i-2*k>=0){
        //当i大于直径时
        //将mid填为i之前的平均值
            arr[mid]=sum/(2*k+1);
            mid++;
            sum-=nums[i-2*k];
        }
    }
    for(int i =mid;i<numsSize;i++){
        arr[i] =-1;
    }
    return arr;
}
```
## 静态填-1
[灵神解答][https://leetcode.cn/problems/k-radius-subarray-averages/solutions/1126276/on-hua-dong-chuang-kou-by-endlesscheng-jtr5/]
使用memset将k+1之前与2k+1之后的内存设置为-1
```
int* getAverages(int* nums, int numsSize, int k, int* returnSize) {
    *returnSize = numsSize;
    int* avgs = malloc(numsSize * sizeof(int));
    memset(avgs, -1, numsSize * sizeof(int));
    long long s = 0; // 维护窗口元素和（注意类型！）
    for (int i = 0; i < numsSize; i++) {
        // 1. 进入窗口
        s += nums[i];
        if (i < k * 2) { // 窗口大小不足 2k+1
            continue;
        }
        // 2. 记录答案
        avgs[i - k] = s / (k * 2 + 1);
        // 3. 离开窗口
        s -= nums[i - k * 2];
    }
    return avgs;
}
```
