# 三指针
![[Pasted image 20260414110429.png]]
```cpp
class Solution {
public:
    int nthUglyNumber(int n) {
        int a = 0, b = 0, c = 0;
		//定义三指针,, 初始abc都指向res[0]
        int res[n];
        res[0] = 1;
        //将res[0]设为1
        //所有丑数都可以/2,/3,/5变成1
        //所以1为最小的丑数
        for(int i = 1; i < n; i++) {
            int n2 = res[a] * 2, n3 = res[b] * 3, n5 = res[c] * 5;
            res[i] = min(min(n2, n3), n5);
            //将当前abc指向的数分别乘2,3,5
            //并将乘完之后最小的丑数放入数组
            if (res[i] == n2) a++;
            if (res[i] == n3) b++;
            if (res[i] == n5) c++;
            //让乘完之后最小的丑数的指针+1
            //其他指针不变
            //运算之后没被填入数的指针一定能成为最小丑数
        }
        return res[n - 1];
    }
};
```
# 小根堆
与仨指针类似, 不人为判断该填入的最小丑数
使用小跟堆的有序性
手动去重
```cpp
class Solution {
public:
    int nthUglyNumber(int n) {
        priority_queue <double,vector<double>,greater<double> > q;
        double answer=1;
        for (int i=1;i<n;++i)
        {
            q.push(answer*2);
            q.push(answer*3);
            q.push(answer*5);
            answer=q.top();
            q.pop();
            while (!q.empty() && answer==q.top()) q.pop();
	        //去重
        }
        return answer;
    }
};
```
