预处理右边的值
左边的值在计算结果时同步算出
```cpp
class Solution {
public:
    long long maximumTripletValue(vector<int>& nums) {
        int n = nums.size();
        vector<int>Rnum(n+1,0);
        long long mid = nums[n-1];
        for(int i = n-2; i>=0;i--){
            Rnum[i] = mid;
            mid = max(mid,(long long)nums[i]);
        }
        //统计右侧可以用于计算的最大值
        long long ret = 0;
        mid = nums[0];
        for(int i =1; i<n;i++){
            if(nums[i]<mid){
                ret = max(ret,(long long)((mid-nums[i])*Rnum[i]));
            }
            mid = max(mid,(long long)nums[i]);
            //此时nums[i] - nums[j]>0
        }
        return ret;
    }
};
//要想(nums[i] - nums[j]) * nums[k]最大
//则nums[i],nums[k]越大越好, nums[j]越小越好
```

# 灵神
一次遍历
记录最大的差值
不用关心左中的顺序
```cpp
class Solution {
public:
    long long maximumTripletValue(vector<int>& nums) {
        long long ans = 0;
        int max_diff = 0, pre_max = 0;
        //max_diff为左中最大的差值,pre_max为左侧最大值 
        for (int x : nums) {
            ans = max(ans, 1LL * max_diff * x);
            max_diff = max(max_diff, pre_max - x);
            pre_max = max(pre_max, x);
        }
        return ans;
    }
};
```
