```c
void LKRever(LK** pphead) {
	if (*pphead != NULL && (*pphead)->next!=NULL)
	//判断为非空链表 和 单节点链表
	{
		LK* left = NULL;
		LK* tail = *pphead;
		LK* right = tail->next;
		while (right != NULL) {
			tail->next = left;
			left = tail;
			tail = right;
			right = tail->next;
		}
		tail->next = left;
		*pphead = tail;
	}
}
```
# 递归法
从链表的尾节点开始，依次向前遍历，遍历过程依次改变各节点的指向
![[0U2343055-7.gif]]![[0U23413F-8.gif]]![[0U234D49-9.gif]]![[0U2345454-10.gif]]
```c
LK* LKRever02(LK** pphead) {
//最终返回新的头结点
	static int fl = 0;
//fl为计数器,标识递归已到最后一层
	if (*pphead == NULL || (*pphead)->next==NULL) {
		return *pphead;
	}
	else
	{
		fl++;
		LK* New_head = LKRever02(&((*pphead)->next));
		//查找最初尾节点
		(*pphead)->next->next = (*pphead);
		(*pphead)->next = NULL;

//fl--;
//当fl==0时,标识递归已到最后一层
//将二级指针指向的一级指针的地址
//由原来的头节点,换成新的头节点

		if (--fl == 0)
		{
			*pphead = New_head;
			return 0;
		}
		return New_head;
	}
}
```
# 头插逆置法
![[0U2342546-11.gif]]![[0U2342R3-12.gif]]![[0U23451H-13.gif]]![[0U234C44-14.gif]]![[0U234C44-14 1.gif]]
```c
int LKRever03(LK** pphead) {
	LK* New_head = NULL;
	LK* temp = NULL;
	if ((*pphead) == NULL||(*pphead)->next == NULL)
	{
		return 0;
	}
	while (*pphead!=NULL) {
		temp = (*pphead)->next;

		(*pphead)->next = New_head;
		New_head = (*pphead);

		*pphead = temp;
	}
	*pphead = New_head;
}
```

```c
int LKRever04(LK** pphead) {
	LK* left = NULL;
	LK* right = NULL;


	if ((*pphead) == NULL || (*pphead)->next == NULL)
	{
		return;
	}

	left = *pphead;
	right = (*pphead)->next;



	while ( right!=NULL ) {
		left->next = right->next;
		right->next = (*pphead);
		*pphead = right;
		right = left->next;
	}
	return 0;

}
//双指针就地头插逆置法
//与头插逆置法类似
//但不新建链表
```
