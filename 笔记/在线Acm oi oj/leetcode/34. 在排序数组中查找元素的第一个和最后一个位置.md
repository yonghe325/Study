[34. 在排序数组中查找元素的第一个和最后一个位置 - 力扣（LeetCode）](https://leetcode.cn/problems/find-first-and-last-position-of-element-in-sorted-array/description/)
# 思路
## 我的思路
+ 要求时间复杂度为 `O(log n)` 的算法解决此问题
	采用二分查找
	分两次查找
	+ 第一次查找target的左边界
		+ 找到左边界:
			以左边界为新的左边界,将右边界重置为最后一个元素
			在查找新的target的右边界
		+ 没找到左边界:
			说明没有target元素,直接跳出
# 解决
 ## 
```c++
class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        vector<int> ret = { -1,-1 };
        if(!(nums.size())){
            return ret;
        }
        //判断长度是否 == 0
        if(nums.size() == 1){
            if(nums[0] == target) ret={0,0};
            return ret;
        }
        //先求左边界
        int L = 0, R = nums.size() - 1, mid = 0;
        while (L!=R) {
            mid = L + ((R - L) >> 1);
            if (nums[mid] >= target) {
                //当nums[mid]>=target时
                //让R向右移
                R = mid;
            }
            else {
                L = mid+1;
            }
        }
        //此时LR重合为左边界
        //找到了,进入寻找右边界的过程
        //没找到,直接跳出
        if (nums[R] == target) {
            ret[0] = R;
            R = nums.size() - 1;
            //左边界就不动了,直接在左边界的右侧寻找右边界
            while (L!=R && L+1 != R) {
                mid = L + ((R - L) >> 1);
                if (nums[mid] <= target) {
                    L = mid;
                }
                else {
                    R = mid;
                }
            }
            //肯定有右边界
            if (nums[R] == target) ret[1] = R;
            else ret[1] = L;
            //当L+1 == R时,要么R是右边界,要么L是右边界
        }
        return ret;
    }
};
```
