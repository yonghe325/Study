```cpp
class Solution {
public:
    int countTrapezoids(vector<vector<int>>& points) {
        map<int,int>arr;
        map<int,int>count;
        long long parallel_count = 0;
        for(int i =0; i<points.size();i++){
            arr[points[i][1]]++;
            parallel_count += arr[points[i][1]]-1;
            count[points[i][1]] += arr[points[i][1]]-1;
        }
        long long ret = 0;
        for(auto i:count ){
            ret += (i.second) * (parallel_count-(i.second));
            parallel_count-=(i.second);
        }
        return ret%(1000000007);
    }
};
// 以y轴作为索引, 存储所有平行于x轴的线
// 然后将所有线段个数向后累积
```

```cpp
class Solution {
public:
    int countTrapezoids(vector<vector<int>>& points) {
        unordered_map<int, int> cnt;  // 统计每个 y 上的点数
        long long total = 0;
        // 统计每个 y 上的点数
        for (auto& p : points) {
            cnt[p[1]]++;
        }
        // 计算所有线段总数
        long long totalSeg = 0;
        vector<long long> segs;
        for (auto& [y, c] : cnt) {
            long long s = (long long)c * (c - 1) / 2;  // C(n,2)求和公式
            segs.push_back(s);
            totalSeg += s;
        }
        // 计算两两组合
        long long ans = 0;
        for (long long s : segs) {
            totalSeg -= s;
            ans += s * totalSeg;
        }
        return ans % 1000000007;
    }
};
```

# 灵神:
```cpp
class Solution {
public:
    int countTrapezoids(vector<vector<int>>& points) {
        const int MOD = 1'000'000'007;
        unordered_map<int, int> cnt;
        for (auto& p : points) {
            cnt[p[1]]++; // 统计每一行（水平线）有多少个点
        }
  
        long long ans = 0, s = 0;
        for (auto& [_, c] : cnt) {
            long long k = 1LL * c * (c - 1) / 2;
            ans += s * k;
            s += k;
        }
        return ans % MOD;
    }
};
```
