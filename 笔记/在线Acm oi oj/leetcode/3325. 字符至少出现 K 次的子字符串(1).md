class Solution {
public:
    int numberOfSubstrings(string s, int k) {
        map<char,int> arr;
        int left = 0;
        int ret = 0;
        for(int i =0;i<s.size();i++){
            arr[s[i]]++;
            while(arr[s[i]] == k){
                arr[s[left]]--;
                left++;
            }
            ret += left;
        }
        return ret;
    }
};
