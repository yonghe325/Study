不定长滑动窗口
```cpp
#define MAX(a, b) a > b ? a : b
class Solution {
public:
    int longestSubarray(vector<int>& nums) {
        int right = 0, left = 0, count = 0, ret = 0;
        for (; right < nums.size(); right++) {
            count += 1 - nums[right];
            // 如果nums[right]为1 , 则count不变
            while (count > 1) {
                count -= 1 - nums[left];
                // 如果nums[left]为0 , 则count-1
                left++;
            }
            ret = MAX(ret, right - left);
        }
        return ret;
    }
};
```
