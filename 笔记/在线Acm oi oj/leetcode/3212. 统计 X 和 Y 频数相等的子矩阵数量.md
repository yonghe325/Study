class Solution {
public:
    int numberOfSubmatrices(vector<vector<char>>& grid) {
        int row = grid.size(), col = grid[0].size();
        vector<vector<vector<int>>>arr(row+1,vector<vector<int>>(col+1,vector<int>(2,0)));
        // [0] 存放X, [1]存放Y
        int ret = 0;
        for(int i = 0; i<row;i++){
            for(int j = 0; j<col; j++){
                arr[i+1][j+1][0] = arr[i+1][j][0] + arr[i][j+1][0] - arr[i][j][0];
                arr[i+1][j+1][1] = arr[i+1][j][1] + arr[i][j+1][1] - arr[i][j][1];
                if(grid[i][j] == 'X'){
                    arr[i+1][j+1][0]++;
                }
                if(grid[i][j] == 'Y'){
                    arr[i+1][j+1][1]++;
                }
                if(arr[i+1][j+1][0] == arr[i+1][j+1][1] && arr[i+1][j+1][0]>0 ){
                    ret++;
                }
            }
        }
        return ret;
    }
};
//三维矩阵
//第三层存放x,y的个数
//统计xy个数时, 计算子矩阵数量

class Solution {
public:
    int numberOfSubmatrices(vector<vector<char>>& grid) {
        int ans = 0, m = grid.size(), n = grid[0].size();
        vector sum(m + 1, vector<array<int, 2>>(n + 1));
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                sum[i + 1][j + 1][0] = sum[i + 1][j][0] + sum[i][j + 1][0] - sum[i][j][0];
                sum[i + 1][j + 1][1] = sum[i + 1][j][1] + sum[i][j + 1][1] - sum[i][j][1];
                if (grid[i][j] != '.') {
                    sum[i + 1][j + 1][grid[i][j] & 1]++;
                    //通过位运算判断当前格子为'X'还是'Y'
                }
                if (sum[i + 1][j + 1][0] && sum[i + 1][j + 1][0] == sum[i + 1][j + 1][1]) {
                    ans++;
                }
            }
        }
        return ans;
    }
};
