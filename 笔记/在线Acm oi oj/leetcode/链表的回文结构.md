# 压栈法
## 后半部分压栈
使用快慢指针查找链表中间结点
```c
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     struct ListNode *next;
 * };
 */
bool isPalindrome(struct ListNode* head) {
    // 使用快慢指针查找链表中间结点
    struct ListNode* fast = head;
    struct ListNode* slow = head;
    while (fast->next && fast->next->next) {
        fast = fast->next->next;
        slow = slow->next;
        // fast指针一次走两步
        // slow指针一次走一步
    }
    // 结束后,slow指针指向中间结点或左侧中间结点
    slow = slow->next;
    // 让slow前进1,指向前一个结点
    // 让右侧压栈
    int arr[100001];
    int index = 0;
    while (slow) {
        arr[index++] = slow->val;
        slow = slow->next;
    }
    // 压栈
    for (int i = index - 1; i >= 0; i--)
    {
        if (head->val == arr[i]) {
            printf("%d %d\n", head->val, arr[i]);
            head = head->next;
        }
        else {
            return false;
        }
    }
    return true;
}
```
## 全部压栈
```c
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     struct ListNode *next;
 * };
 */

bool isPalindrome(struct ListNode* head) {
    //哑节点
    struct ListNode* dummy = head;
    // 使用数组实现栈
    // int* arr = (int*)malloc(sizeof(int));
    int* arr = (int*)malloc(sizeof(int)*0);
    //跟随原链表压栈
    int index = 0;
    while(head){
        arr = (int*)realloc(arr,sizeof(int)*(index+1));
        arr[index++] = head->val;
        head=head->next;
    }
    //完成压栈
    //index为数组长度
    // 遍历原链表,顺便出栈比较
    head = dummy;
    for(int i = index-1 ; i>=0;i--)
    {
        if(head->val == arr[i]){
            printf("%d %d\n",head->val,arr[i]);
            head=head->next;
        }
        else{
            return false;
        }
    }
    return true;
}
```

# 翻转中间链表 
 ```c
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     struct ListNode *next;
 * };
 */
bool isPalindrome(struct ListNode* head) {
    //求长度
    //从中间到尾节点逆序
    //分别从头,和尾向中间遍历
        //相同返回t
        //不同返回f
    struct ListNode* dummy = (struct ListNode*)malloc(sizeof(struct ListNode));
    dummy->next = head;
    struct ListNode* tail = dummy;
    int len = 0;
    while(tail->next !=NULL){
        tail=tail->next  ;
        len++;
    }
    if(len == 1){
        return true;
    }
    // if(len == 2 && head->val == head->next->val){
    //     return true;
    // }
    // if(len == 2 && head->val != head->next->val){
    //     return false;
    // }

    struct ListNode* start  =dummy;
    for(int i =0;i<(len/2);i++ ){
        start=start->next;
    }

    //找到中间节点
    struct ListNode* deg = start;
    struct ListNode* mid = deg->next;
    // struct ListNode* end = mid->next;
    tail = mid->next;
    for(int i =0;i<(len/2)+1;i++){
            mid->next = deg;
            if(!tail){
                break;
            }
            else{
                deg =mid;
                mid = tail;
                tail = tail->next;
            }
    }
    tail =mid;
    start = dummy->next;

    for(int i =0;i<(len/2);i++){
        if(start->val != tail->val){
            return false;
        }
        else{
            start=start->next;
            tail=tail->next;
        }
    }
    return true;
}
 ```
