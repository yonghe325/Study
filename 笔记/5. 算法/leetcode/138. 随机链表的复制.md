# 后缀法
```c
/**
 * Definition for a Node.
 * struct Node {
 *     int val;
 *     struct Node *next;
 *     struct Node *random;
 * };
 */
struct Node* copyRandomList(struct Node* head) {
    if (head == NULL) {
        return NULL;
    }
    struct Node* dummy = (struct Node*)malloc(sizeof(struct Node));
    dummy->next=head;
    // struct Node* insert = (struct Node*)malloc(sizeof(struct Node));
    while(head){
        struct Node* insert = (struct Node*)malloc(sizeof(struct Node));
        insert->val=head->val;
        insert->next = head->next;
        // insert->random = NULL;
        head->next = insert;
        head=head->next->next;
        //head跳过插入的结点
    }
    //完成插入
    //连线结点
    head = dummy->next;
    while(head){
        head->next->random = head->random?head->random->next :NULL;
        //实现连线
        head=head->next->next;
    }
    //连线完毕
    //分离结点
    head = dummy->next;
    struct Node* New_head = head->next;
    struct Node* mid = NULL;
    while(head->next->next){
        mid = head->next;
        head->next = mid->next;
        mid->next = mid->next->next;
        head = head->next;
    }
    head->next = NULL;
    return New_head;
  
}
```
