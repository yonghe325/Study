通过迭代,先让前i个有序
时间复杂度
$$O(N^2)$$
```
int InsertionSort(int *arr,int length) {
	if (!arr || length < 2) {
		return 1;
	}

	for (size_t i = 1; i < length; i++)
	{
		for(int j = i ; j >0 ; j--)
		//j从右向左移动
		{
			if (arr[j] < arr[j-1])
			{
				//遇到比arr[j]大的数
				//二者换位
				Swap(arr, j, j - 1);
			}
			else if (arr[j] >= arr[j - 1]) {
				//遇到比arr[j]小的数
				//由于前i个已经有序
				//arr[j]不会再遇到比自己大的数
				//直接break
				break;
			}
		}
	}
	return 0;
}
```
