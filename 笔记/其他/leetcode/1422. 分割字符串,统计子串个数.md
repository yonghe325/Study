本题为滑动窗口的基础
本质是元素的移入移出
# 分割法 
先遍历一遍统计右边的个数
随后隔板向右移动
移动时判断,
当移入左边满足的字符,左边+1;
当右边满足的字符移出右边,右边-1;
```
#define MAX(a, b) ((b) > (a) ? (b) : (a))
int maxScore(char* s) {
    int len =strlen(s);
    //先统计1的个数
    int R = 0;
    for(int i = 0;i<len;i++ ){
        R+= (s[i] == '1')?1:0;
    }
    
    int L =0;
    //移动中计算L
    int sum =0;
    //计算总和
    
    //移动分割线
    for(int i =0;s[i+1];i++){
    //隔板最多移动到倒数第二个
        if( s[i] == '0' ){//左边规定字符移入左边,左边++
            L++;
        }
        else{
        //由于就两种字符,不是左边规定字符,就是右边规定字符
        //右边规定字符移入左边,右边--
            R--;
        }
        sum = MAX(sum,(L+R));
    }
    return sum;
}
```
