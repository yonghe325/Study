# 滑动窗口
# if分支法
```
#define MIN(a,b) ((a<b)?(a):(b))
int minimumRecolors(char* blocks, int k) {
    int mid =0;
    int ret =k+1;
    for(int i =0;blocks[i];i++){
        if(blocks[i] == 'W'){
            mid++;
        }
        if(i-k+1 < 0){
            continue;
        }
        ret =MIN(ret,mid);
        if(blocks[i-k+1] == 'W'){
            mid--;
        }
    }
    return ret;
}
```
# 位运算法
由于在ASCLL码中
W= 01010111
B = 01000010
所以可以使用&判断是否为W
```
int minimumRecolors(char* blocks, int k) {
    int mid =0;
    for(int i= 0;i<k;i++){
        mid+=blocks[i] & 1;
    }

    int ret =mid;

    for(int i = k;i<strlen(blocks);i++){
        mid += ((blocks[i] & 1) -( blocks[i-k] &1));
        ret = MIN(ret,mid);
    }
    return ret;
}
```
